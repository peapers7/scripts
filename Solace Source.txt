then            return value        elseif type(value) == "string" then            local ok, result = pcall(Font.fromName, value, weight)            if ok then                return result            else                print("SetFont: Font family '" .. value .. "' not found, keeping previous font")                return nil            end        end        return nil    end    if type(fontConfig) == "table" then        if fontConfig.Regular then            local resolved = resolveFont(fontConfig.Regular, Enum.FontWeight.Medium)            if resolved then                Solace.CustomFont = resolved            end        end        if fontConfig.Bold then            local resolved = resolveFont(fontConfig.Bold, Enum.FontWeight.Bold)            if resolved then                Solace.CustomFontBold = resolved            end        end    elseif type(fontConfig) == "string" or typeof(fontConfig) == "Font" then        local resolvedRegular = resolveFont(fontConfig, Enum.FontWeight.Medium)        local resolvedBold = resolveFont(fontConfig, Enum.FontWeight.Bold)        if resolvedRegular then            Solace.CustomFont = resolvedRegular        end        if resolvedBold then            Solace.CustomFontBold = resolvedBold        end    else        print("SetFont: Invalid font configuration. Use a string, a Font datatype, or a table with Regular/Bold keys")    endendfunction Solace.Instance(class, props)    local inst = Instance.new(class)    if inst:IsA("GuiBase") then        inst.AutoLocalize = false    end    local parent = nil    for k, v in pairs(props or {}) do        if k == "Parent" then            parent = v        else            inst[k] = v        end    end    if parent then        inst.Parent = parent    end    return instendfunction Solace.RegisterSliderElement(element)    table.insert(Solace.SliderElements, element)endfunction Solace.ValidateResizeSize(width, height)    local minWidth = 400    local minHeight = 300    local maxWidth = 800    local maxHeight = 400    width = math.clamp(width, minWidth, maxWidth)    height = math.clamp(height, minHeight, maxHeight)    return width, heightendfunction Solace.SaveResizeSettings()    local ok, err = pcall(function()        local json = HttpService:JSONEncode({            Width = Solace.SavedResize.Width,            Height = Solace.SavedResize.Height        })        safeEnsureFolder(Solace.ConfigFolder)        safeWriteFile(Solace.ConfigFolder .. Use only letters, numbers, spaces, underscores and hyphens"    end    local configData = {        Elements = {},        ConfigName = name,        SaveTime = os.time()    }    for _, elementData in ipairs(Solace.SaveableElements or {}) do        local element = elementData.Element        local value = elementData.GetValue and elementData.GetValue() or nil        if value ~= nil and not Solace.SaveExclusions[element.Name] then            local elementInfo = {                Type = element.Type,                Name = element.Name,                TabName = elementData.TabName,                Value = value            }            table.insert(configData.Elements, elementInfo)        end    end    local success, result = pcall(function()        local json = HttpService:JSONEncode(configData)        safeEnsureFolder(Solace.ConfigFolder)        safeWriteFile(Solace.ConfigFolder .. "/" .. name .. ".json", json)        return true    end)    if success and result then        Solace.RefreshConfigList()        return true    else        print("Failed to save config '" .. name .. "':", result or "Unknown error")        return false, result or "Error saving configuration"    endendfunction Solace.LoadConfig(name)    if not name or name == "" then        print("Cannot load config: No name provided")        return false, "Config name cannot be empty"    end    local json = safeReadFile(Solace.ConfigFolder .. "/" .. name .. ".json")    if not json then        print("Failed to load config '" .. name .. "': file not found or unreadable")        return false, "Configuration file not found"    end    local ok, configData = pcall(HttpService.JSONDecode, HttpService, json)    if not ok or not configData or not configData.Elements then        print("Failed to parse config '" .. name .. "'")        return false, "Invalid configuration file format"    end    local loadedCount = 0    for _, elementInfo in ipairs(configData.Elements) do        for _, elementData in ipairs(Solace.SaveableElements or {}) do            local element = elementData.Element            if element.Type == elementInfo.Type and element.Name == elementInfo.Name and elementData.TabName == elementInfo.TabName then                if not elementData.IsAutoLoadInput then                    if element.Type == "Toggle" or element.Type == "Slider" or element.Type == "Dropdown" or element.Type == "Input" or element.Type == "Checkbox" then                        pcall(elementData.SetValue, elementInfo.Value)                        loadedCount = loadedCount + 1                    end                end            end        end    end    Solace.CurrentConfig = name    if loadedCount == 0 then        print("Config '" .. name .. "' loaded but no settings were applied")    end    return trueendfunction Solace.DeleteConfig(name)    if not name or name == "" then        print("Cannot delete config: No name provided")        return false, "Config name cannot be empty"    end    local ok = safeDeleteFile(Solace.ConfigFolder ..
